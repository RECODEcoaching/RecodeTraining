SQL requis — si tu ne vois pas les séances, exécute dans Supabase → SQL Editor : 
CREATE TABLE IF NOT EXISTS seance_exercices (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  seance_id   UUID REFERENCES seances(id) ON DELETE CASCADE,
  exercice_id UUID REFERENCES exercices(id) ON DELETE SET NULL,
  nom_exo     TEXT NOT NULL,
  ordre       INTEGER DEFAULT 0,
  series      INTEGER DEFAULT 3,
  reps        TEXT DEFAULT '10',
  charge      TEXT,
  rir         INTEGER DEFAULT 2,
  repos_serie INTEGER DEFAULT 90,
  repos_cote  INTEGER,
  unilateral  BOOLEAN DEFAULT false,
  notes       TEXT,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE seance_exercices ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all anon on seance_exercices"
  ON seance_exercices FOR ALL TO anon USING (true) WITH CHECK (true);
SQL requis — pour activer la section "Échauffement" dans les séances, exécute dans Supabase → SQL Editor : 
CREATE TABLE IF NOT EXISTS seance_echauffement (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  seance_id   UUID REFERENCES seances(id) ON DELETE CASCADE,
  exercice_id UUID REFERENCES exercices(id) ON DELETE SET NULL,
  nom_exo     TEXT NOT NULL,
  ordre       INTEGER DEFAULT 0,
  consigne    TEXT,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE seance_echauffement ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all anon on seance_echauffement"
  ON seance_echauffement FOR ALL TO anon USING (true) WITH CHECK (true);
